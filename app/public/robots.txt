# DinoDreamer. Crawlers may read the top-level pages. Every page can open a panel for each of the
# 16,384 basis groups (/basis/<group>, /index/basis/<group> and so on), and one panel loads about 70
# library files. The panels, their old /block/ addresses, the paged and filtered views (an address with
# a query) and the owner pages stay out of crawls (gallery/cloudflare-settings.md).
User-agent: *
Disallow: /basis/
Disallow: /*/basis/
Disallow: /block/
Disallow: /*/block/
Disallow: /*?
Disallow: /admin

# library reads. X and LinkedIn obey robots.txt: without this group, a shared /automata?group=... or /basis/...
# link would show no card there. A crawler obeys only the group that names it.
User-agent: Twitterbot
User-agent: LinkedInBot
Disallow: /admin
